🚀 EMPEZAR EN 3 PASOS
═══════════════════════════════════════════════════════════════════════════════════

PASO 1: Abre una relation manager
────────────────────────────────────
  Ejemplo: app/Filament/Resources/Parties/RelationManagers/PartyRolesRelationManager.php

PASO 2: Agrega el trait (2 líneas)
────────────────────────────────────
  use App\Filament\Traits\HasIconizedActions;  // ← Agregar al inicio
  
  class PartyRolesRelationManager extends RelationManager
  {
      use HasIconizedActions;  // ← Agregar aquí
      // ...
  }

PASO 3: Envuelve las acciones (1 cambio por acción)
────────────────────────────────────────────────────
  // ANTES:                          // DESPUÉS:
  EditAction::make()                 $this->withActionIcons(EditAction::make(), 'edit')
  DeleteAction::make()               $this->withActionIcons(DeleteAction::make(), 'delete')
  CreateAction::make()               $this->withActionIcons(CreateAction::make(), 'create')


📝 EJEMPLO COMPLETO
═══════════════════════════════════════════════════════════════════════════════════

<?php

namespace App\Filament\Resources\Parties\RelationManagers;

use App\Filament\Traits\HasIconizedActions;  // ← Agregar
use App\Filament\Resources\PartyRoles\Schemas\PartyRoleForm;
use App\Filament\Resources\PartyRoles\Tables\PartyRolesTable;
use Filament\Actions\CreateAction;
use Filament\Actions\DeleteAction;
use Filament\Actions\EditAction;
use Filament\Resources\RelationManagers\RelationManager;
use Filament\Schemas\Schema;
use Filament\Tables\Table;

class PartyRolesRelationManager extends RelationManager
{
    use HasIconizedActions;  // ← Agregar

    protected static string $relationship = 'partyRoles';

    public function form(Schema $schema): Schema
    {
        return PartyRoleForm::configure($schema);
    }

    public function table(Table $table): Table
    {
        return PartyRolesTable::configure($table)
            ->recordActions([
                $this->withActionIcons(EditAction::make(), 'edit'),      // ← Cambiar
                $this->withActionIcons(DeleteAction::make(), 'delete'),  // ← Cambiar
            ]);
    }
}

🎯 CAMBIAR UN ICONO (SUPER FÁCIL)
═══════════════════════════════════════════════════════════════════════════════════

Edita: config/action-icons.php

'edit' => [
    'icon' => 'tabler-pencil',  ← CAMBIA AQUÍ  
    'tooltip' => 'Editar',
],

✨ El cambio se aplica AUTOMÁTICAMENTE a todas las relation managers.
